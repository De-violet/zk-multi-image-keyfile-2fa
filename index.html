1. Instalasi SDK Klien:

            
npm install zk-canvas-sdk


            
2. Inisialisasi Canvas (Klien):

            
import { ZkCanvasSDK } from 'zk-canvas-sdk';

const sdk = new ZkCanvasSDK({
  canvas: document.getElementById('totpCanvas'),
  secret: 'USER_MASTER_SECRET',
  rotationIntervalMs: 35000 // Siklus 35 detik
});

// Dengarkan event waktu & perubahan matriks
sdk.on('tick', ({ remainingMs }) => console.log(remainingMs));
sdk.start();

// Hitung bukti Groth16 di background Worker
const { proof, publicSignals } = await sdk.generateProof(sessionNonce);


            
3. Verifikasi Server (Express REST API):

            
import { verifyVisualTotpProof } from './zkVerifier.js';

app.post('/api/auth/verify-2fa', async (req, res) => {
  const { proof, publicSignals, clientTimeWindow, sessionNonce } = req.body;
  const result = await verifyVisualTotpProof({
    proof,
    publicSignals,
    clientTimeWindow,
    sessionNonce,
    intervalSec: 35
  });
  if (!result.valid) return res.status(401).json({ error: result.error });
  return res.json({ success: true, token: createToken() });
});
